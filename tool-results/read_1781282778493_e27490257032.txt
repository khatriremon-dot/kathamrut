   160→// ─── Sortable Row ───────────────────────────────────────────────────────────
   161→function SortableChapterRow({
   162→  chapter,
   163→  onEdit,
   164→  onDelete,
   165→}: {
   166→  chapter: ChapterRow;
   167→  onEdit: (ch: ChapterRow) => void;
   168→  onDelete: (ch: ChapterRow) => void;
   169→}) {
   170→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({
   171→    id: chapter.id,
   172→  });
   173→  const style = {
   174→    transform: CSS.Transform.toString(transform),
   175→    transition,
   176→    opacity: isDragging ? 0.5 : 1,
   177→  };
   178→
   179→  return (
   180→    <TableRow ref={setNodeRef} style={style}>
   181→      <TableCell className="w-10">
   182→        <button {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing p-1 text-muted-foreground hover:text-foreground">
   183→          <GripVertical className="w-4 h-4" />
   184→        </button>
   185→      </TableCell>
   186→      <TableCell className="font-medium">{chapter.number}</TableCell>
   187→      <TableCell className="font-medium">{chapter.title}</TableCell>
   188→      <TableCell className="text-muted-foreground text-sm">
   189→        {chapter.content.split(/\s+/).filter(Boolean).length.toLocaleString()} words
   190→      </TableCell>
   191→      <TableCell className="text-muted-foreground text-sm">
   192→        {new Date(chapter.createdAt).toLocaleDateString()}
   193→      </TableCell>
   194→      <TableCell>
   195→        <div className="flex items-center gap-1">
   196→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => onEdit(chapter)}>
   197→            <Pencil className="w-3.5 h-3.5" />
   198→          </Button>
   199→          <Button variant="ghost" size="icon" className="h-8 w-8 text-destructive" onClick={() => onDelete(chapter)}>
   200→            <Trash2 className="w-3.5 h-3.5" />
   201→          </Button>
   202→        </div>
   203→      </TableCell>
   204→    </TableRow>
   205→  );
   206→}

   210→// Authenticated fetch helper - reads stored key from localStorage
   211→function authFetch(options: RequestInit = {}): RequestInit {
   212→  let adminKey = '';
   213→  if (typeof window !== 'undefined') {
   214→    adminKey = localStorage.getItem('kathamrut_admin_auth') || '';
   215→  }
   216→  return {
   217→    ...options,
   218→    headers: {
   219→      ...options.headers,
   220→      'x-admin-key': adminKey,
   221→    },
   222→  };
   223→}

   225→export default function AdminPage() {
   226→  const [isAuthenticated, setIsAuthenticated] = useState(() => {
   227→    if (typeof window !== 'undefined') {
   228→      const storedAuth = localStorage.getItem('kathamrut_admin_auth');
   229→      return !!storedAuth;
   230→    }
   231→    return false;
   232→  });
   233→  const [passwordInput, setPasswordInput] = useState('');
   234→  const [showPasswordError, setShowPasswordError] = useState(false);
   235→
   236→  const [loginError, setLoginError] = useState('');
   237→
   238→  const handleLogin = async () => {
   239→    // Verify password server-side so it's never in the client bundle
   240→    setLoginError('');
   241→    try {
   242→      const res = await fetch('/api/auth/verify', {
   243→        method: 'POST',
   244→        headers: {
   245→          'Content-Type': 'application/json',
   246→          'x-admin-key': passwordInput,
   247→        },
   248→      });
   249→      const data = await res.json();
   250→      if (res.ok) {
   251→        localStorage.setItem('kathamrut_admin_auth', passwordInput);
   252→        setIsAuthenticated(true);
   253→        setShowPasswordError(false);
   254→      } else {
   255→        setShowPasswordError(true);
   256→        setLoginError(data.error || 'Login failed');
   257→      }
   258→    } catch (err) {
   259→      setShowPasswordError(true);
   260→      setLoginError('Network error — could not reach server');
   261→    }
   262→  };
   263→
   264→  const handleLogout = () => {
   265→    localStorage.removeItem('kathamrut_admin_auth');
   266→    setIsAuthenticated(false);
   267→    setPasswordInput('');
   268→  };
   269→
   270→  // Password gate
   271→  if (!isAuthenticated) {
   272→    return (
   273→      <div className="min-h-screen bg-background flex items-center justify-center">
   274→        <div className="w-full max-w-sm space-y-6 p-8">
   275→          <div className="text-center">
   276→            <div className="w-16 h-16 rounded-2xl bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center mx-auto mb-4">
   277→              <Lock className="w-8 h-8 text-amber-600" />
   278→            </div>
   279→            <h1 className="text-2xl font-bold">Admin Access</h1>
   280→            <p className="text-sm text-muted-foreground mt-1">Enter the admin password to continue</p>
   281→          </div>
   282→          <div className="space-y-3">
   283→            <Input
   284→              type="password"
   285→              placeholder="Password"
   286→              value={passwordInput}
   287→              onChange={(e) => { setPasswordInput(e.target.value); setShowPasswordError(false); }}
   288→              onKeyDown={(e) => { if (e.key === 'Enter') handleLogin(); }}
   289→              autoFocus
   290→            />
   291→            {showPasswordError && (
   292→              <div className="text-sm text-destructive space-y-1">
   293→                <p>Incorrect password. Please try again.</p>
   294→                {loginError && <p className="text-xs opacity-75">Server: {loginError}</p>}
   295→              </div>
   296→            )}
   297→            <Button onClick={handleLogin} className="w-full bg-amber-600 hover:bg-amber-700 text-white">
   298→              Sign In
   299→            </Button>
   300→          </div>
   301→          <a
   302→            href="/"
   303→            className="block text-center text-sm text-muted-foreground hover:text-foreground transition-colors"
   304→          >
   305→            ← Back to Site
   306→          </a>
   307→        </div>
   308→      </div>
   309→    );
   310→  }
   311→
   312→  return (
   313→    <div className="min-h-screen bg-background">
   314→      {/* Header */}
   315→      <header className="sticky top-0 z-30 bg-background/80 backdrop-blur-md border-b border-border">

   322→            <BookOpen className="w-5 h-5 text-amber-600" />
   323→            <h1 className="font-bold text-lg">Admin Dashboard</h1>
   324→          </div>
   325→          <Button variant="ghost" size="sm" className="ml-auto gap-2 text-muted-foreground hover:text-destructive" onClick={handleLogout}>
   326→            <LogOut className="w-4 h-4" />
   327→            <span className="hidden sm:inline">Logout</span>
   328→          </Button>
   329→        </div>
   330→      </header>

   375→function NovelsTab() {
   376→  const [novels, setNovels] = useState<NovelRow[]>([]);
   377→  const [loading, setLoading] = useState(true);
   378→  const [dialogOpen, setDialogOpen] = useState(false);
   379→  const [deleteOpen, setDeleteOpen] = useState(false);
   380→  const [editingNovel, setEditingNovel] = useState<NovelRow | null>(null);
   381→  const [deletingNovel, setDeletingNovel] = useState<NovelRow | null>(null);
   382→  const [saving, setSaving] = useState(false);
   383→
   384→  // Form state
   385→  const [formTitle, setFormTitle] = useState('');
   386→  const [formAuthor, setFormAuthor] = useState('');
   387→  const [formDescription, setFormDescription] = useState('');
   388→  const [formCoverUrl, setFormCoverUrl] = useState('');
   389→  const [formLanguage, setFormLanguage] = useState('en');
   390→  const [formCategory, setFormCategory] = useState('fiction');
   391→  const [formRating, setFormRating] = useState('4.5');
   392→  const [formStatus, setFormStatus] = useState('ongoing');
   393→  const [uploading, setUploading] = useState(false);
   394→
   395→  const fetchNovels = useCallback(async () => {
   396→    setLoading(true);
   397→    try {
   398→      const res = await fetch('/api/novels');
   399→      const data = await res.json();
   400→      setNovels(data);
   401→    } catch {
   402→      toast.error('Failed to load novels');
   403→    } finally {
   404→      setLoading(false);
   405→    }
   406→  }, []);
   407→
   408→  useEffect(() => { fetchNovels(); }, [fetchNovels]);
   409→
   410→  const resetForm = () => {
   411→    setFormTitle('');
   412→    setFormAuthor('');
   413→    setFormDescription('');
   414→    setFormCoverUrl('');
   415→    setFormLanguage('en');
   416→    setFormCategory('fiction');
   417→    setFormRating('4.5');
   418→    setFormStatus('ongoing');
   419→    setEditingNovel(null);
   420→  };
   421→
   422→  const openCreate = () => {
   423→    resetForm();
   424→    setDialogOpen(true);
   425→  };
   426→
   427→  const openEdit = (novel: NovelRow) => {
   428→    setEditingNovel(novel);
   429→    setFormTitle(novel.title);
   430→    setFormAuthor(novel.author);
   431→    setFormDescription(novel.description);
   432→    setFormCoverUrl(novel.coverUrl);
   433→    setFormLanguage(novel.language);
   434→    setFormCategory(novel.category);
   435→    setFormRating(String(novel.rating));
   436→    setFormStatus(novel.status);
   437→    setDialogOpen(true);
   438→  };
   439→
   440→  const handleSave = async () => {
   441→    if (!formTitle.trim() || !formAuthor.trim()) {
   442→      toast.error('Title and Author are required');
   443→      return;
   444→    }
   445→    setSaving(true);
   446→    try {
   447→      const body = {
   448→        title: formTitle,
   449→        author: formAuthor,
   450→        description: formDescription,
   451→        coverUrl: formCoverUrl,
   452→        language: formLanguage,
   453→        category: formCategory,
   454→        rating: parseFloat(formRating),
   455→        status: formStatus,
   456→      };
   457→
   458→      if (editingNovel) {
   459→        const res = await fetch(`/api/novels/${editingNovel.id}`, authFetch({
   460→          method: 'PUT',
   461→          headers: { 'Content-Type': 'application/json' },
   462→          body: JSON.stringify(body),
   463→        }));
   464→        if (!res.ok) throw new Error('Update failed');
   465→        toast.success('Novel updated');
   466→      } else {
   467→        const res = await fetch('/api/novels', authFetch({
   468→          method: 'POST',
   469→          headers: { 'Content-Type': 'application/json' },
   470→          body: JSON.stringify(body),
   471→        }));
   472→        if (!res.ok) throw new Error('Create failed');
   473→        toast.success('Novel created');
   474→      }
   475→      setDialogOpen(false);
   476→      fetchNovels();
   477→    } catch {
   478→      toast.error('Failed to save novel');
   479→    } finally {
   480→      setSaving(false);
   481→    }
   482→  };
   483→
   484→  const handleDelete = async () => {
   485→    if (!deletingNovel) return;
   486→    setSaving(true);
   487→    try {
   488→      const res = await fetch(`/api/novels/${deletingNovel.id}`, authFetch({ method: 'DELETE' }));
   489→      if (!res.ok) throw new Error('Delete failed');
   490→      toast.success('Novel deleted');
   491→      setDeleteOpen(false);
   492→      fetchNovels();
   493→    } catch {
   494→      toast.error('Failed to delete novel');
   495→    } finally {
   496→      setSaving(false);
   497→      setDeletingNovel(null);
   498→    }
   499→  };
   500→
   501→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   502→    const file = e.target.files?.[0];
   503→    if (!file) return;
   504→    setUploading(true);
   505→    try {
   506→      const fd = new FormData();
   507→      fd.append('file', file);
   508→      const res = await fetch('/api/upload', authFetch({ method: 'POST', body: fd }));
   509→      const data = await res.json();
   510→      if (data.url) {
   511→        setFormCoverUrl(data.url);
   512→        toast.success('Image uploaded');
   513→      } else {
   514→        toast.error(data.error || 'Upload failed');
   515→      }
   516→    } catch {
   517→      toast.error('Upload failed');
   518→    } finally {
   519→      setUploading(false);
   520→    }
   521→  };
   522→
   523→  return (
   524→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   525→      <div className="flex items-center justify-between mb-4">
   526→        <h2 className="text-xl font-bold">Novels</h2>
   527→        <Button onClick={openCreate} className="gap-2 bg-amber-600 hover:bg-amber-700 text-white">
   528→          <Plus className="w-4 h-4" />
   529→          Add Novel
   530→        </Button>
   531→      </div>
   532→
   533→      {loading ? (
   534→        <div className="space-y-3">
   535→          {[...Array(5)].map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}
   536→        </div>
   537→      ) : novels.length === 0 ? (
   538→        <div className="text-center py-12 text-muted-foreground">
   539→          <BookOpen className="w-10 h-10 mx-auto mb-3 opacity-30" />
   540→          <p>No novels yet. Create your first novel!</p>
   541→        </div>
   542→      ) : (
   543→        <div className="border rounded-lg overflow-hidden">
   544→          <div className="max-h-[500px] overflow-y-auto custom-scrollbar">
   545→            <Table>
   546→              <TableHeader>
   547→                <TableRow>
   548→                  <TableHead>Title</TableHead>
   549→                  <TableHead className="hidden sm:table-cell">Author</TableHead>
   550→                  <TableHead className="hidden md:table-cell">Language</TableHead>
   551→                  <TableHead className="hidden md:table-cell">Category</TableHead>
   552→                  <TableHead className="hidden lg:table-cell">Status</TableHead>
   553→                  <TableHead className="hidden lg:table-cell">Chapters</TableHead>
   554→                  <TableHead className="text-right">Actions</TableHead>
   555→                </TableRow>
   556→              </TableHeader>
   557→              <TableBody>
   558→                {novels.map((novel) => (
   559→                  <TableRow key={novel.id}>
   560→                    <TableCell className="font-medium max-w-[200px] truncate">{novel.title}</TableCell>
   561→                    <TableCell className="hidden sm:table-cell text-muted-foreground">{novel.author}</TableCell>
   562→                    <TableCell className="hidden md:table-cell">
   563→                      <Badge variant="secondary" className="text-xs">{LANGUAGE_LABELS[novel.language] || novel.language}</Badge>
   564→                    </TableCell>
   565→                    <TableCell className="hidden md:table-cell">
   566→                      <Badge variant="outline" className="text-xs capitalize">{novel.category}</Badge>
   567→                    </TableCell>
   568→                    <TableCell className="hidden lg:table-cell">
   569→                      <Badge variant="secondary" className={`text-xs ${novel.status === 'completed' ? 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400' : 'bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-400'}`}>
   570→                        {novel.status}
   571→                      </Badge>
   572→                    </TableCell>
   573→                    <TableCell className="hidden lg:table-cell text-muted-foreground">{novel.chapters.length}</TableCell>
   574→                    <TableCell>
   575→                      <div className="flex items-center justify-end gap-1">
   576→                        <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => { window.location.hash = '#chapters'; window.dispatchEvent(new CustomEvent('admin-view-chapters', { detail: novel.id })); }}>
