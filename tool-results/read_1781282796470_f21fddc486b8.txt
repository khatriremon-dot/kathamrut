   210→   210→// Authenticated fetch helper - reads stored key from localStorage
   211→   211→function authFetch(options: RequestInit = {}): RequestInit {
   212→   212→  let adminKey = '';
   213→   213→  if (typeof window !== 'undefined') {
   214→   214→    adminKey = localStorage.getItem('kathamrut_admin_auth') || '';
   215→   215→  }
   216→   216→  return {
   217→   217→    ...options,
   218→   218→    headers: {
   219→   219→      ...options.headers,
   220→   220→      'x-admin-key': adminKey,
   221→   221→    },
   222→   222→  };
   223→   223→}

   233→   233→  const [passwordInput, setPasswordInput] = useState('');

   239→   239→    // Verify password server-side so it's never in the client bundle
   240→   240→    setLoginError('');
   241→   241→    try {
   242→   242→      const res = await fetch('/api/auth/verify', {
   243→   243→        method: 'POST',
   244→   244→        headers: {
   245→   245→          'Content-Type': 'application/json',
   246→   246→          'x-admin-key': passwordInput,
   247→   247→        },
   248→   248→      });
   249→   249→      const data = await res.json();
   250→   250→      if (res.ok) {
   251→   251→        localStorage.setItem('kathamrut_admin_auth', passwordInput);
   252→   252→        setIsAuthenticated(true);
   253→   253→        setShowPasswordError(false);
   254→   254→      } else {
   255→   255→        setShowPasswordError(true);
   256→   256→        setLoginError(data.error || 'Login failed');
   257→   257→      }
   258→   258→    } catch (err) {
   259→   259→      setShowPasswordError(true);
   260→   260→      setLoginError('Network error — could not reach server');
   261→   261→    }

   265→   265→    localStorage.removeItem('kathamrut_admin_auth');
